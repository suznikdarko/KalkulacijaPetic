import customtkinter as ctk

ctk.set_appearance_mode("dark")
ctk.set_default_color_theme("blue")

def izracun():
    try:
        naklada = int(entry_naklada.get())
        papir = float(entry_papir.get())
        barve = entry_barve.get()

        material = naklada * papir / 1000

        b = barve.split("/")
        plosce = int(b[0]) + int(b[1]) - 1
        cena_plosc = plosce * 10

        priprava = 25

        skupaj = material + cena_plosc + priprava

        rezultat.configure(text=f"{round(skupaj,2)} €")

    except:
        rezultat.configure(text="Napaka")

app = ctk.CTk()
app.title("TISK PRO APP")
app.geometry("400x400")

title = ctk.CTkLabel(app, text="TISK KALKULATOR", font=("Arial", 20))
title.pack(pady=20)

entry_naklada = ctk.CTkEntry(app, placeholder_text="Naklada")
entry_naklada.pack(pady=10)

entry_papir = ctk.CTkEntry(app, placeholder_text="Cena papirja / 1000")
entry_papir.pack(pady=10)

entry_barve = ctk.CTkEntry(app, placeholder_text="Barve (4/4)")
entry_barve.pack(pady=10)

btn = ctk.CTkButton(app, text="IZRAÈUNAJ", command=izracun)
btn.pack(pady=20)

rezultat = ctk.CTkLabel(app, text="", font=("Arial", 18))
rezultat.pack(pady=20)

app.mainloop()
